cmake_minimum_required(VERSION 3.24)
project(gpu_performance_lab LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(GPU_LAB_BUILD_CUDA "Build PyTorch CUDA extension modules" OFF)

if(GPU_LAB_BUILD_CUDA)
  enable_language(CUDA)
  set(CMAKE_CUDA_STANDARD 17)
  set(CMAKE_CUDA_STANDARD_REQUIRED ON)
  find_package(Python3 REQUIRED COMPONENTS Interpreter Development.Module)
  execute_process(COMMAND "${Python3_EXECUTABLE}" -c
    "import torch; print(torch.utils.cmake_prefix_path)"
    OUTPUT_VARIABLE TORCH_PREFIX OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
  list(APPEND CMAKE_PREFIX_PATH "${TORCH_PREFIX}")
  find_package(Torch REQUIRED)
  find_package(CUDAToolkit REQUIRED)
  find_library(TORCH_PYTHON_LIBRARY torch_python PATHS "${TORCH_INSTALL_PREFIX}/lib" REQUIRED)
  separate_arguments(TORCH_CXX_FLAGS_LIST NATIVE_COMMAND "${TORCH_CXX_FLAGS}")

  function(gpu_lab_extension workload)
    set(target "gpu_lab_${workload}")
    Python3_add_library(${target} MODULE WITH_SOABI bindings.cpp naive.cu optimized.cu)
    target_include_directories(${target} PRIVATE "${PROJECT_SOURCE_DIR}/common/cuda" ${TORCH_INCLUDE_DIRS})
    target_link_libraries(${target} PRIVATE ${TORCH_LIBRARIES} ${TORCH_PYTHON_LIBRARY} CUDA::cudart)
    target_compile_definitions(${target} PRIVATE TORCH_EXTENSION_NAME=${target})
    target_compile_options(${target} PRIVATE ${TORCH_CXX_FLAGS_LIST} -O3
      $<$<COMPILE_LANGUAGE:CUDA>:-lineinfo>)
    set_target_properties(${target} PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/python")
  endfunction()
  add_subdirectory(kernels/01_vector_add/cuda)
  add_subdirectory(kernels/02_reduction/cuda)
  add_subdirectory(kernels/03_softmax/cuda)
else()
  message(STATUS "CUDA extensions disabled; enable GPU_LAB_BUILD_CUDA on an NVIDIA CUDA host")
endif()
